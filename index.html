function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); // Jika ada parameter 'saveData' di URL, berarti HTML sedang mencoba menyimpan data! if (e.parameter.saveData) { try { var payload = e.parameter.saveData; sheet.getRange("A1").setValue(payload); return ContentService.createTextOutput(JSON.stringify({status: "success"})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({status: "error"})) .setMimeType(ContentService.MimeType.JSON); } } // Jika tidak ada parameter 'saveData', berarti HTML sedang membaca/polling data var data = sheet.getRange("A1").getValue(); if (!data) data = JSON.stringify({ status: "empty", message: "Belum ada data" }); return ContentService.createTextOutput(data).setMimeType(ContentService.MimeType.JSON); } // Fungsi POST opsional jika suatu saat dibutuhkan function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); try { var payload = e.postData.contents; sheet.getRange("A1").setValue(payload); return ContentService.createTextOutput(JSON.stringify({status: "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
